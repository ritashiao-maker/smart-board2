        // ==========================================
        // 4. TDX OAuth 2.0 Token 與即時 API 串接
        // ==========================================
        async function getTDXAccessToken() {
            if (!state.clientId || !state.clientSecret) {
                throw new Error('未設定 TDX 金鑰');
            }
            if (state.accessToken && Date.now() < state.tokenExpiresAt - 60000) {
                return state.accessToken;
            }

            const tokenUrl = 'https://tdx.transportdata.tw/auth/realms/TDXConnect/protocol/openid-connect/token';
            const bodyParams = new URLSearchParams();
            bodyParams.append('grant_type', 'client_credentials');
            bodyParams.append('client_id', state.clientId);
            bodyParams.append('client_secret', state.clientSecret);

            const response = await fetch(tokenUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
                body: bodyParams.toString()
            });

            if (!response.ok) {
                throw new Error(`TDX 驗證失敗 (HTTP ${response.status})`);
            }

            const tokenJson = await response.json();
            if (!tokenJson.access_token) {
                throw new Error('無法取得 TDX Token');
            }

            state.accessToken = tokenJson.access_token;
            state.tokenExpiresAt = Date.now() + (tokenJson.expires_in || 86400) * 1000;
            return state.accessToken;
        }

        async function fetchTDXJson(url, token) {
            const res = await fetch(url, {
                headers: {
                    'Authorization': `Bearer ${token}`,
                    'Accept': 'application/json'
                }
            });
            if (!res.ok) throw new Error(`API 異常 (${res.status})`);
            return await res.json();
        }

        async function refreshAllTransitData(isManual = false) {
            const icon = document.getElementById('manual-refresh-icon');
            icon.classList.add('animate-spin');

            const statusDot = document.getElementById('tdx-status-dot');
            const statusText = document.getElementById('tdx-status-text');

            if (state.clientId && state.clientSecret) {
                try {
                    const token = await getTDXAccessToken();
                    await fetchRealTDXData(token);
                    state.isLiveTDX = true;
                    statusDot.className = 'inline-block w-2.5 h-2.5 rounded-full bg-emerald-400 live-indicator';
                    statusText.innerText = 'TDX 官方即時連線中（臺中市北區全區動態）';
                    if (isManual) showToast('已同步交通部 TDX 最新即時運輸動態！');
                } catch (err) {
                    state.isLiveTDX = false;
                    generateRealisticBackupData();
                    statusDot.className = 'inline-block w-2.5 h-2.5 rounded-full bg-amber-400 live-indicator';
                    statusText.innerText = `備援模式（TDX 異常：${err.message}）`;
                    if (isManual) showToast('TDX 連線異常，已自動切換為即時備援數據');
                }
            } else {
                state.isLiveTDX = false;
                generateRealisticBackupData();
                statusDot.className = 'inline-block w-2.5 h-2.5 rounded-full bg-amber-400 live-indicator';
                statusText.innerText = '即時備援模式（免金鑰運作中）';
                if (isManual) showToast('已更新臺中市北區即時運輸動態');
            }

            renderAllModules();
            state.countdown = REFRESH_INTERVAL_SEC;
            setTimeout(() => icon.classList.remove('animate-spin'), 500);
        }

        function buildShareableUrl() {
            // 強制使用您的 GitHub 正式公開網址作為 QR Code 核心基底
            const baseUrl = "https://ritashiao-maker.github.io/smart-board/";
            const url = new URL(baseUrl);
            url.searchParams.set('stop', state.selectedBusStop);
            if (state.onlyLowFloor) url.searchParams.set('acc', '1');
            if (state.isLargeFont) url.searchParams.set('lg', '1');
            return url.toString();
        }

                state.themeMode = localStorage.getItem(STORAGE_KEYS.themeMode) || 'auto';
                state.isLargeFont = localStorage.getItem(STORAGE_KEYS.largeFont) === 'true';

                const savedMarquee = localStorage.getItem(STORAGE_KEYS.marqueeText);
                document.getElementById('marquee-display').innerText =
                    (savedMarquee && savedMarquee.trim()) ? savedMarquee : DEFAULT_MARQUEE;
            } catch (e) {}

            // 解析網址參數 (手機掃碼開啟時自動套用)
            const params = new URLSearchParams(window.location.search);
            if (params.get('stop')) state.selectedBusStop = params.get('stop');
            if (params.get('acc') === '1') state.onlyLowFloor = true;
            if (params.get('lg') === '1') state.isLargeFont = true;

            // 同步 UI 按鈕狀態
            applyThemeAndAccessibility();
            if (state.onlyLowFloor) {
                document.getElementById('filter-lowfloor').classList.add('bg-sky-500', 'text-slate-950');
            }
            document.querySelectorAll('.stop-tab').forEach(btn => {
                if (btn.getAttribute('data-stop') === state.selectedBusStop) {
                    btn.className = 'stop-tab active px-3 py-1.5 rounded-xl text-xs sm:text-sm font-black bg-sky-500 text-slate-950 shadow transition';
                } else {
                    btn.className = 'stop-tab ctrl-btn px-3 py-1.5 rounded-xl text-xs sm:text-sm font-bold transition';
                }
            });

            // 啟動時鐘與資料刷新
            tickClockAndCountdown();
            setInterval(tickClockAndCountdown, 1000);
            refreshAllTransitData(false);

            // 綁定無障礙與友善工具列按鈕
            document.getElementById('btn-voice-read').addEventListener('click', speakNextArrivals);

            document.getElementById('btn-font-scale').addEventListener('click', () => {
                state.isLargeFont = !state.isLargeFont;
                try { localStorage.setItem(STORAGE_KEYS.largeFont, String(state.isLargeFont)); } catch (e) {}
                applyThemeAndAccessibility();
                showToast(state.isLargeFont ? '已切換為「長輩大字體模式」' : '已恢復「標準字體大小」');
            });

            document.getElementById('btn-theme-cycle').addEventListener('click', () => {
                const order = ['auto', 'day', 'night', 'hc'];
                const nextIdx = (order.indexOf(state.themeMode) + 1) % order.length;
                state.themeMode = order[nextIdx];
                try { localStorage.setItem(STORAGE_KEYS.themeMode, state.themeMode); } catch (e) {}
                applyThemeAndAccessibility();
                const labels = { auto: '自動日夜切換', day: '日間明亮模式', night: '夜間護眼模式', hc: '視障高對比模式' };
                showToast(`已切換視覺風格：${labels[state.themeMode]}`);
            });

            // 公車站牌篩選
            document.querySelectorAll('.stop-tab').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.stop-tab').forEach(b => {
                        b.className = 'stop-tab ctrl-btn px-3 py-1.5 rounded-xl text-xs sm:text-sm font-bold transition';
                    });
                    btn.className = 'stop-tab active px-3 py-1.5 rounded-xl text-xs sm:text-sm font-black bg-sky-500 text-slate-950 shadow transition';
                    state.selectedBusStop = btn.getAttribute('data-stop');
                    renderBusList();
                });
            });

            // 低地板無障礙篩選
            document.getElementById('filter-lowfloor').addEventListener('click', (e) => {
                state.onlyLowFloor = !state.onlyLowFloor;
                const btn = e.currentTarget;
                if (state.onlyLowFloor) {
                    btn.classList.add('bg-sky-500', 'text-slate-950');
                    btn.classList.remove('bg-sky-500/15', 'text-sky-400');
                } else {
                    btn.classList.remove('bg-sky-500', 'text-slate-950');
                    btn.classList.add('bg-sky-500/15', 'text-sky-400');
                }
                renderBusList();
            });

            // 婦幼安心站篩選
            document.getElementById('filter-safestop').addEventListener('click', (e) => {
                state.onlySafeStop = !state.onlySafeStop;
                const btn = e.currentTarget;
                if (state.onlySafeStop) {
                    btn.classList.add('bg-pink-500', 'text-white');
                    btn.classList.remove('bg-pink-500/15', 'text-pink-400');
                } else {
                    btn.classList.remove('bg-pink-500', 'text-white');
                    btn.classList.add('bg-pink-500/15', 'text-pink-400');
                }
                applyThemeAndAccessibility();
                renderBusList();
            });

            // 臺鐵車站切換
            document.querySelectorAll('.tra-tab').forEach(btn => {
                btn.addEventListener('click', () => {
                    document.querySelectorAll('.tra-tab').forEach(b => {
                        b.className = 'tra-tab px-2.5 py-1 rounded-lg text-xs font-bold text-secondary hover:text-primary transition';
                    });
                    btn.className = 'tra-tab active px-2.5 py-1 rounded-lg text-xs font-black bg-blue-500 text-white transition';
                    state.selectedTraStation = btn.getAttribute('data-tra');
                    renderTRA();
                });
            });

            // 手動刷新
            document.getElementById('btn-manual-refresh').addEventListener('click', () => refreshAllTransitData(true));

            // 手機帶著走 QR Code Modal
            const modalQr = document.getElementById('modal-qr');
            const closeQr = () => { modalQr.classList.add('hidden'); modalQr.classList.remove('flex'); };
            document.getElementById('btn-open-qr').addEventListener('click', openMobileQRModal);
            document.getElementById('btn-close-qr').addEventListener('click', closeQr);
            document.getElementById('btn-done-qr').addEventListener('click', closeQr);
            document.getElementById('btn-copy-url').addEventListener('click', () => {
                const urlText = document.getElementById('qr-url-preview').innerText;
                const tempInput = document.createElement('input');
                tempInput.value = urlText;
                document.body.appendChild(tempInput);
                tempInput.select();
                try {
                    document.execCommand('copy');
                    showToast('已複製看板連結，可貼給家人或傳至手機！');
                } catch (err) {
                    showToast('請長按上方網址手動複製');
                }
                document.body.removeChild(tempInput);
            });

            // TDX API Modal
            const modalTdx = document.getElementById('modal-tdx');
            const inputId = document.getElementById('input-tdx-id');
            const inputSecret = document.getElementById('input-tdx-secret');
            const closeTdx = () => { modalTdx.classList.add('hidden'); modalTdx.classList.remove('flex'); };

            document.getElementById('btn-open-tdx').addEventListener('click', () => {
                inputId.value = state.clientId;
                inputSecret.value = state.clientSecret;
                modalTdx.classList.remove('hidden');
                modalTdx.classList.add('flex');
            });
            document.getElementById('btn-close-tdx').addEventListener('click', closeTdx);
            document.getElementById('btn-cancel-tdx').addEventListener('click', closeTdx);
            document.getElementById('btn-save-tdx').addEventListener('click', () => {
                const newId = inputId.value.trim();
                const newSecret = inputSecret.value.trim();
                if (!newId || !newSecret) {
                    showToast('請完整輸入 TDX Client ID 與 Client Secret');
                    return;
                }
                state.clientId = newId;
                state.clientSecret = newSecret;
                state.accessToken = '';
                state.tokenExpiresAt = 0;
                try {
                    localStorage.setItem(STORAGE_KEYS.clientId, newId);
                    localStorage.setItem(STORAGE_KEYS.clientSecret, newSecret);
                } catch (e) {}
                closeTdx();
                refreshAllTransitData(true);
            });
            document.getElementById('btn-clear-tdx').addEventListener('click', () => {
                inputId.value = '';
                inputSecret.value = '';
                state.clientId = '';
                state.clientSecret = '';
                state.accessToken = '';
                state.tokenExpiresAt = 0;
                try {
                    localStorage.removeItem(STORAGE_KEYS.clientId);
                    localStorage.removeItem(STORAGE_KEYS.clientSecret);
                } catch (e) {}
                closeTdx();
                refreshAllTransitData(true);
            });

            // 跑馬燈編輯 Modal
            const modalMarquee = document.getElementById('modal-marquee');
            const textareaMarquee = document.getElementById('textarea-marquee');
            const closeMarquee = () => { modalMarquee.classList.add('hidden'); modalMarquee.classList.remove('flex'); };

            document.getElementById('btn-edit-marquee').addEventListener('click', () => {
                textareaMarquee.value = document.getElementById('marquee-display').innerText;
                modalMarquee.classList.remove('hidden');
                modalMarquee.classList.add('flex');
            });
            document.getElementById('btn-close-marquee').addEventListener('click', closeMarquee);
            document.getElementById('btn-cancel-marquee').addEventListener('click', closeMarquee);
            document.getElementById('btn-reset-marquee').addEventListener('click', () => {
                textareaMarquee.value = DEFAULT_MARQUEE;
            });
            document.getElementById('btn-save-marquee').addEventListener('click', () => {
                const val = textareaMarquee.value.trim() || DEFAULT_MARQUEE;
                try { localStorage.setItem(STORAGE_KEYS.marqueeText, val); } catch (e) {}
                document.getElementById('marquee-display').innerText = val;
                closeMarquee();
                showToast('北區交通與友善快訊已更新！');
            });
        });
    </script>
